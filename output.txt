import json, sys
sys.path.insert(0, "scripts"); sys.path.insert(0, ".")
from collections import defaultdict
from koz3.score import score
from koz3.nested import weighted, _add_all, _add
from sp_oof_local import region_folds

names = json.load(open("work/rehearsal_sp/names.json"))     # anon → real region
RF = region_folds()                                         # real region → fold
fold_of = {a: RF[r] for a, r in names.items() if r in RF}

gt = defaultdict(list)
for f in json.load(open("work/rehearsal_sp/gt.geojson"))["features"]:
    gt[f["properties"]["region_name"]].append(f)

def counts(path):
    d = json.load(open(path)); P = defaultdict(list)
    for f in d["features"]:
        P[f["properties"]["region_name"]].append(f)
    out = {}
    for reg, gts in gt.items():
        _, res = score({"type": "FeatureCollection", "features": P.get(reg, [])},
                       {"type": "FeatureCollection", "features": gts})
        out[reg] = {c: (v["tp"], v["fp"], v["fn"]) for c, v in res.items()}
    return out

C = {k: counts(f"work/rehearsal_sp/result_{p}.geojson") for k, p in
     (("base", "base"), ("rho0.2", "probe_0.2_0.02"), ("rho0.35", "probe_0.35_0.02"))}
folds = sorted({fold_of[a] for a in C["base"] if a in fold_of})
print("folds:", {f: sum(1 for a in C['base'] if fold_of.get(a) == f) for f in folds})

for k in ("rho0.2", "rho0.35"):
    print(f"  {k:7s} pooled {weighted(_add_all(C[k].values())):.4f}")

picks, tot = {}, {}
for f in folds:
    pick = max(("rho0.2", "rho0.35"),
               key=lambda k: weighted(_add_all(C[k][a] for a in C[k] if fold_of.get(a) != f)))
    picks[f] = pick
    tot = _add(tot, _add_all(C[pick][a] for a in C[pick] if fold_of.get(a) == f))
print("chosen per fold:", picks)
print(f"HONEST (nested) score {weighted(tot):.4f}")
