# 4. zip from the tree ROOT (no wrapper folder), files only
./myenv/bin/python - <<'EOF'
import os, zipfile
root, zp = "build/rebuild1", "build/rebuild1.zip"
with zipfile.ZipFile(zp, "w", zipfile.ZIP_DEFLATED) as z:
    for dp, dns, fs in os.walk(root):
        dns.sort()
        for f in sorted(fs):
            p = os.path.join(dp, f)
            z.write(p, os.path.relpath(p, root))
print(zp, round(os.path.getsize(zp) / 1e9, 3), "GB")
EOF
